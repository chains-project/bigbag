package org.example.migration;

import spoon.Launcher;
import spoon.processing.AbstractProcessor;
import spoon.reflect.code.CtExpression;
import spoon.reflect.code.CtFieldRead;
import spoon.reflect.code.CtInvocation;
import spoon.reflect.code.CtTypeAccess;
import spoon.reflect.declaration.CtMethod;
import spoon.reflect.factory.Factory;
import spoon.reflect.reference.CtFieldReference;
import spoon.reflect.reference.CtTypeReference;
import spoon.support.sniper.SniperJavaPrettyPrinter;

import java.util.List;

public class MinaSslRefactoring {

    /**
     * Processor to handle the removal of SslFilter.initiateHandshake(IoSession).
     * Refactoring strategy:
     * Old: sslFilter.initiateHandshake(session);
     * New: ((SslHandler) session.getAttribute(SslFilter.SSL_HANDLER)).handshake();
     */
    public static class SslHandshakeProcessor extends AbstractProcessor<CtInvocation<?>> {

        @Override
        public boolean isToBeProcessed(CtInvocation<?> candidate) {
            // 1. Check Method Name
            if (!"initiateHandshake".equals(candidate.getExecutable().getSimpleName())) {
                return false;
            }

            // 2. Check Argument Count (We target the 1-arg version primarily)
            List<CtExpression<?>> args = candidate.getArguments();
            if (args.size() != 1) {
                return false;
            }

            // 3. Defensive Type Checks (NoClasspath support)
            // Check Owner: Should be SslFilter
            CtExpression<?> target = candidate.getTarget();
            if (target != null) {
                CtTypeReference<?> targetType = target.getType();
                if (targetType != null && !targetType.getQualifiedName().contains("SslFilter")) {
                    return false;
                }
            }

            // Check Argument: Should be IoSession
            CtExpression<?> arg = args.get(0);
            CtTypeReference<?> argType = arg.getType();
            if (argType != null && !argType.getQualifiedName().contains("IoSession")) {
                return false;
            }

            // If we are here, it's a likely match.
            return true;
        }

        @Override
        public void process(CtInvocation<?> invocation) {
            Factory factory = getFactory();
            CtExpression<?> sessionArg = invocation.getArguments().get(0);

            // Create References for Types
            CtTypeReference<?> sslFilterRef = factory.Type().createReference("org.apache.mina.filter.ssl.SslFilter");
            CtTypeReference<?> sslHandlerRef = factory.Type().createReference("org.apache.mina.filter.ssl.SslHandler");
            CtTypeReference<?> objectRef = factory.Type().objectType();

            // 1. Create access to SslFilter.SSL_HANDLER
            // Code: SslFilter.SSL_HANDLER
            CtFieldReference<?> sslHandlerFieldRef = factory.Field().createReference(
                    sslFilterRef, 
                    objectRef, 
                    "SSL_HANDLER"
            );
            sslHandlerFieldRef.setStatic(true);
            
            CtTypeAccess<?> sslFilterAccess = factory.Code().createTypeAccess(sslFilterRef);
            CtFieldRead<?> sslHandlerFieldRead = factory.Code().createFieldRead();
            sslHandlerFieldRead.setTarget(sslFilterAccess);
            sslHandlerFieldRead.setVariable(sslHandlerFieldRef);

            // 2. Create invocation: session.getAttribute(SslFilter.SSL_HANDLER)
            CtInvocation<?> getAttributeInvocation = factory.Code().createInvocation(
                    sessionArg.clone(),
                    factory.Method().createReference(
                            sessionArg.getType() != null ? sessionArg.getType() : factory.Type().createReference("org.apache.mina.core.session.IoSession"),
                            objectRef,
                            "getAttribute",
                            objectRef // Argument type
                    ),
                    sslHandlerFieldRead
            );

            // 3. Create Cast: (SslHandler) ...
            CtExpression<?> castedExpression = factory.Code().createTypeCast(sslHandlerRef, getAttributeInvocation);

            // 4. Create final invocation: .handshake()
            CtInvocation<?> handshakeInvocation = factory.Code().createInvocation(
                    castedExpression,
                    factory.Method().createReference(
                            sslHandlerRef,
                            factory.Type().voidPrimitiveType(),
                            "handshake"
                    )
            );

            // Replace the original invocation
            invocation.replace(handshakeInvocation);
            
            System.out.println("Refactored SslFilter.initiateHandshake at " + invocation.getPosition());
        }
    }

    public static void main(String[] args) {
        String inputPath = "./src/main/java";
        String outputPath = "./src/main/gen";

        Launcher launcher = new Launcher();
        launcher.addInputResource(inputPath);
        launcher.setSourceOutputDirectory(outputPath);

        // CRITICAL SETTINGS for robust refactoring
        launcher.getEnvironment().setCommentEnabled(true);
        launcher.getEnvironment().setPrettyPrinterCreator(
            () -> new SniperJavaPrettyPrinter(launcher.getEnvironment())
        );
        launcher.getEnvironment().setNoClasspath(true);

        launcher.addProcessor(new SslHandshakeProcessor());

        try {
            launcher.run();
            System.out.println("Refactoring complete. Output in: " + outputPath);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
