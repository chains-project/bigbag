package org.example.migration;

import spoon.Launcher;
import spoon.processing.AbstractProcessor;
import spoon.reflect.code.CtInvocation;
import spoon.reflect.code.CtExpression;
import spoon.reflect.reference.CtTypeReference;
import spoon.support.sniper.SniperJavaPrettyPrinter;

public class ReflectionsRefactoring {

    /**
     * Processor to migrate org.reflections usage from Guava Predicates (apply) 
     * to Java 8 Predicates (test).
     * 
     * Affected classes include:
     * - org.reflections.util.FilterBuilder
     * - org.reflections.ReflectionUtils (return values)
     * - org.reflections.vfs.UrlTypeVFS
     * - org.reflections.vfs.Vfs
     */
    public static class ApplyToTestProcessor extends AbstractProcessor<CtInvocation<?>> {
        @Override
        public boolean isToBeProcessed(CtInvocation<?> candidate) {
            // 1. Name Check: We are looking for 'apply'
            if (!"apply".equals(candidate.getExecutable().getSimpleName())) {
                return false;
            }

            // 2. Argument Check: Predicate.test(T) takes exactly 1 argument
            if (candidate.getArguments().size() != 1) {
                return false;
            }

            // 3. Type/Owner Check: Determine if the target object belongs to org.reflections
            CtExpression<?> target = candidate.getTarget();
            if (target == null) return false;

            // Strategy A: Check the resolved type of the expression (e.g. variable type)
            CtTypeReference<?> targetType = target.getType();
            if (targetType != null) {
                String typeName = targetType.getQualifiedName();
                // Filter out QueryFunction as it explicitly ADDED apply(Object) in the diff
                if (typeName.contains("org.reflections") && !typeName.contains("QueryFunction")) {
                    return true;
                }
            }

            // Strategy B: If target is a method invocation (e.g. ReflectionUtils.withName("x").apply(y))
            // we check the declaring class of that method invocation.
            if (target instanceof CtInvocation) {
                CtInvocation<?> chainedInv = (CtInvocation<?>) target;
                CtTypeReference<?> declaringType = chainedInv.getExecutable().getDeclaringType();
                if (declaringType != null) {
                    String ownerName = declaringType.getQualifiedName();
                    // ReflectionUtils returns Predicates that now use 'test' instead of 'apply'
                    if (ownerName.contains("org.reflections") && !ownerName.contains("QueryFunction")) {
                        return true;
                    }
                }
            }

            return false;
        }

        @Override
        public void process(CtInvocation<?> invocation) {
            // Transformation: Rename 'apply' to 'test'
            invocation.getExecutable().setSimpleName("test");
            System.out.println("Refactored 'apply' to 'test' at line " + invocation.getPosition().getLine());
        }
    }

    public static void main(String[] args) {
        // Default paths (can be modified by user)
        String inputPath = "./src/main/java";
        String outputPath = "./src/main/gen";

        Launcher launcher = new Launcher();
        launcher.addInputResource(inputPath);
        launcher.setSourceOutputDirectory(outputPath);

        // CRITICAL: Configure Sniper Printer for source code preservation
        launcher.getEnvironment().setCommentEnabled(true);
        launcher.getEnvironment().setPrettyPrinterCreator(
            () -> new SniperJavaPrettyPrinter(launcher.getEnvironment())
        );
        // Defensive: Handle missing classpath gracefully
        launcher.getEnvironment().setNoClasspath(true);

        launcher.addProcessor(new ApplyToTestProcessor());

        try {
            launcher.run();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
